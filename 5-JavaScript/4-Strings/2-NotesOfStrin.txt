# JAVASCRIPT STRINGS - COMPLETE GUIDE
# ===================================

## 1. STRING FUNDAMENTALS
## =======================

### What are Strings?
Strings are sequences of characters used to represent text. In JavaScript, strings are primitive values but have object-like properties and methods.

### Key Characteristics:
- IMMUTABLE: Cannot be changed once created
- PRIMITIVE type but have object-like methods
- Zero-based indexing
- Can use single quotes ('), double quotes ("), or backticks (`)
- UTF-16 encoded

### String Creation:
let str1 = 'Hello';          // Single quotes
let str2 = "World";          // Double quotes  
let str3 = `Hello World`;    // Template literals
let str4 = String(123);      // String constructor
let str5 = new String("Hi"); // String object (avoid)

## 2. STRING CREATION METHODS
## ==========================

### Literal Creation:
let single = 'Single quotes';
let double = "Double quotes";
let template = `Template literals`;

### Constructor:
let fromNumber = String(42);        // "42"
let fromBoolean = String(true);     // "true"
let fromObject = String({});        // "[object Object]"

### Template Literals (ES6):
let name = "John";
let age = 30;
let message = `Hello ${name}, you are ${age} years old`;
// "Hello John, you are 30 years old"

### Multi-line Strings:
// Old way (with escape characters)
let oldMulti = "Line 1\nLine 2\nLine 3";

// Template literals (cleaner)
let newMulti = `Line 1
Line 2
Line 3`;

## 3. STRING PROPERTIES
## ====================

### length Property:
let str = "Hello";
console.log(str.length); // 5

### Accessing Characters:
let str = "Hello";

console.log(str[0]);     // "H" - bracket notation
console.log(str.charAt(0)); // "H" - charAt method
console.log(str.at(-1)); // "o" - negative indexing (ES2022)

### Bracket vs charAt():
let str = "Hello";

console.log(str[10]);     // undefined
console.log(str.charAt(10)); // "" (empty string)

## 4. STRING MANIPULATION METHODS
## ==============================

### Concatenation:
let str1 = "Hello";
let str2 = "World";

let result1 = str1 + " " + str2;          // "Hello World"
let result2 = str1.concat(" ", str2);     // "Hello World"
let result3 = `${str1} ${str2}`;          // "Hello World"

### Case Conversion:
let str = "Hello World";

console.log(str.toUpperCase());    // "HELLO WORLD"
console.log(str.toLowerCase());    // "hello world"
console.log("hello".toUpperCase()); // "HELLO"

### Trimming Whitespace:
let str = "   Hello World   ";

console.log(str.trim());        // "Hello World"
console.log(str.trimStart());   // "Hello World   "
console.log(str.trimEnd());     // "   Hello World"

### Padding:
let str = "5";

console.log(str.padStart(3, "0")); // "005"
console.log(str.padEnd(3, "0"));   // "500"

## 5. STRING SEARCHING METHODS
## ===========================

### indexOf() & lastIndexOf():
let str = "Hello World, Hello Universe";

console.log(str.indexOf("Hello"));     // 0
console.log(str.indexOf("Hello", 1));  // 13 (start from index 1)
console.log(str.lastIndexOf("Hello")); // 13
console.log(str.indexOf("Hi"));        // -1 (not found)

### includes(), startsWith(), endsWith():
let str = "Hello World";

console.log(str.includes("World"));    // true
console.log(str.startsWith("Hello"));  // true  
console.log(str.endsWith("World"));    // true
console.log(str.startsWith("World", 6)); // true (start from index 6)

### search() with Regex:
let str = "Hello World";

console.log(str.search(/world/i));     // 6 (case-insensitive)
console.log(str.search("World"));      // 6

## 6. STRING EXTRACTION METHODS
## =============================

### slice():
let str = "Hello World";

console.log(str.slice(0, 5));    // "Hello" (end not included)
console.log(str.slice(6));       // "World"
console.log(str.slice(-5));      // "World" (from end)
console.log(str.slice(3, 7));    // "lo W"

### substring():
let str = "Hello World";

console.log(str.substring(0, 5)); // "Hello"
console.log(str.substring(6));    // "World"

// Difference: substring() swaps if start > end
console.log(str.substring(5, 0)); // "Hello" (swaps parameters)
console.log(str.slice(5, 0));     // "" (empty)

### substr() [Deprecated]:
// Avoid using - deprecated
console.log(str.substr(0, 5));    // "Hello" (start, length)

### split():
let str = "apple,banana,orange";
let fruits = str.split(",");      // ["apple", "banana", "orange"]

let words = "Hello World".split(" "); // ["Hello", "World"]
let chars = "Hello".split("");    // ["H", "e", "l", "l", "o"]

## 7. STRING REPLACEMENT METHODS
## =============================

### replace():
let str = "Hello World";

console.log(str.replace("World", "Universe")); // "Hello Universe"

// Only replaces first occurrence
let text = "cat dog cat";
console.log(text.replace("cat", "bird")); // "bird dog cat"

### replaceAll() (ES2021):
let text = "cat dog cat";
console.log(text.replaceAll("cat", "bird")); // "bird dog bird"

### Regex Replacement:
let str = "Hello World";

// Replace all occurrences with regex
console.log(str.replace(/l/g, "x")); // "Hexxo Worxd"

// Case-insensitive replacement
console.log(str.replace(/hello/i, "Hi")); // "Hi World"

## 8. TEMPLATE LITERALS (ES6)
## ==========================

### Basic Interpolation:
let name = "John";
let age = 30;
let message = `Hello ${name}, you are ${age} years old`;

### Expressions:
let a = 5;
let b = 10;
console.log(`Five plus ten is ${a + b}`); // "Five plus ten is 15"

### Multi-line Strings:
let html = `
<div>
    <h1>Title</h1>
    <p>Content</p>
</div>
`;

### Tagged Templates:
function highlight(strings, ...values) {
    return strings.reduce((result, str, i) => {
        return result + str + (values[i] ? `<mark>${values[i]}</mark>` : '');
    }, '');
}

let name = "John";
let age = 30;
let result = highlight`Hello ${name}, you are ${age} years old`;
// "Hello <mark>John</mark>, you are <mark>30</mark> years old"

## 9. STRING COMPARISON METHODS
## ============================

### localeCompare():
let str1 = "apple";
let str2 = "banana";

console.log(str1.localeCompare(str2)); // -1 (apple comes before banana)
console.log(str2.localeCompare(str1)); // 1 (banana comes after apple)
console.log(str1.localeCompare(str1)); // 0 (equal)

### Case-sensitive Comparison:
console.log("hello" === "Hello"); // false
console.log("hello" === "hello"); // true

### Case-insensitive Comparison:
console.log("hello".toLowerCase() === "Hello".toLowerCase()); // true

## 10. UNICODE AND CHARACTER METHODS
## =================================

### charCodeAt() & codePointAt():
let str = "Hello";

console.log(str.charCodeAt(0));  // 72 (UTF-16 code)
console.log(str.codePointAt(0)); // 72 (Unicode code point)

### fromCharCode() & fromCodePoint():
console.log(String.fromCharCode(72));  // "H"
console.log(String.fromCodePoint(72)); // "H"

### Unicode Escape Sequences:
let str = "\u0048\u0065\u006C\u006C\u006F"; // "Hello"
let heart = "\u2764"; // "❤"

## 11. REGULAR EXPRESSIONS WITH STRINGS
## ====================================

### test() and match():
let str = "Hello World";

console.log(/world/i.test(str)); // true
console.log(str.match(/world/i)); // ["World"]

### matchAll() (ES2020):
let str = "test1 test2 test3";
let matches = [...str.matchAll(/test(\d)/g)];

matches.forEach(match => {
    console.log(match[0], match[1]); // test1 1, test2 2, test3 3
});

### split() with Regex:
let str = "apple, banana; orange";
let fruits = str.split(/[,;]\s*/); // ["apple", "banana", "orange"]

## 12. STRING VALIDATION METHODS
## =============================

### Checking Empty Strings:
let str = "";

console.log(str.length === 0);        // true
console.log(str === "");              // true
console.log(!str);                    // true (falsy check)

### Checking Whitespace:
let str = "   ";

console.log(str.trim().length === 0); // true
console.log(/^\s*$/.test(str));       // true

### Validation Patterns:
let email = "test@example.com";

console.log(email.includes("@"));                  // Basic check
console.log(/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)); // Regex validation

## 13. PERFORMANCE CONSIDERATIONS
## ==============================

### String Immutability:
// Inefficient - creates new strings each time
let str = "Hello";
for (let i = 0; i < 1000; i++) {
    str += "!"; // Creates new string each iteration
}

// Better - use array join
let parts = ["Hello"];
for (let i = 0; i < 1000; i++) {
    parts.push("!");
}
let result = parts.join("");

### Memory Efficiency:
- Small strings: Use literals
- Large strings: Consider Array.join() for building
- Template literals: Good for readability, similar performance

## 14. STRING ITERATION METHODS
## ============================

### for...of Loop:
let str = "Hello";

for (let char of str) {
    console.log(char); // H, e, l, l, o
}

### Array.from():
let str = "Hello";
let chars = Array.from(str); // ["H", "e", "l", "l", "o"]

### Spread Operator:
let str = "Hello";
let chars = [...str]; // ["H", "e", "l", "l", "o"]

### for Loop:
for (let i = 0; i < str.length; i++) {
    console.log(str[i]);
}

## 15. REAL-WORLD STRING PATTERNS
## ==============================

### URL Manipulation:
let url = "https://example.com/products?id=123";

// Extract query parameters
let params = url.split("?")[1]?.split("&") || [];
let query = {};
params.forEach(param => {
    let [key, value] = param.split("=");
    query[key] = value;
});

### Text Processing:
let text = "   Hello, WORLD!   ";

// Clean and format text
let cleaned = text
    .trim()
    .toLowerCase()
    .replace(/[^a-z\s]/g, "") // Remove non-alphabet characters
    .replace(/\s+/g, " ");    // Normalize spaces

console.log(cleaned); // "hello world"

### Template Processing:
function formatMessage(template, data) {
    return template.replace(/\${(\w+)}/g, (match, key) => {
        return data[key] || match;
    });
}

let message = formatMessage("Hello ${name}, welcome to ${city}!", {
    name: "John",
    city: "New York"
});
// "Hello John, welcome to New York!"

## 16. COMMON PITFALLS AND SOLUTIONS
## ==================================

### 1. Mutability Misunderstanding:
let str = "Hello";
str[0] = "J"; // Doesn't work - strings are immutable
console.log(str); // "Hello"

// Solution: Create new string
let newStr = "J" + str.slice(1); // "Jello"

### 2. Encoding Issues:
// Special characters
console.log("café".length); // 4 (correct)
console.log("👍".length);   // 2 (surrogate pair)

// Solution: Use Array.from() for character count
console.log(Array.from("👍").length); // 1

### 3. Case-sensitive Operations:
let input = "YES";
if (input === "yes") { // false
    // Won't execute
}

// Solution: Normalize case
if (input.toLowerCase() === "yes") { // true
    // Will execute
}

### 4. Whitespace Handling:
let input = "  hello  ";
console.log(input === "hello"); // false

// Solution: Trim before comparison
console.log(input.trim() === "hello"); // true

## 18. BEST PRACTICES
## ==================

### 1. Use Template Literals for Dynamic Strings:
// ❌ Old way
let message = "Hello " + name + ", you are " + age + " years old";

// ✅ Modern way
let message = `Hello ${name}, you are ${age} years old`;

### 2. Prefer const for String Variables:
// ✅ Good
const GREETING = "Hello";
const API_URL = "https://api.example.com";

// ❌ Avoid
let greeting = "Hello";

### 3. Use Descriptive Names:
// ✅ Good
const USER_WELCOME_MESSAGE = "Welcome to our application!";
const MAX_USERNAME_LENGTH = 50;

// ❌ Avoid
const str = "Welcome";
const num = 50;

### 4. Handle Internationalization:
// Consider locale-aware methods
console.log("straße".localeCompare("strasse", "de")); // German locale
