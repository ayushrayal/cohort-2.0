## 1. FUNCTION DECLARATION (HOISTED)
====================================
function functionName(parameters) {
    // code to be executed
    return value;
}

// Example:
function greet(name) {
    return "Hello " + name;
}
console.log(greet("Buddy")); // "Hello Buddy"

## 2. FUNCTION EXPRESSION (NOT HOISTED)
======================================
const functionName = function(parameters) {
    // code
};

// Example:
const square = function(x) {
    return x * x;
};
console.log(square(5)); // 25

## 3. ARROW FUNCTIONS (ES6)
==========================
const functionName = (parameters) => {
    // code
};

// Examples:
const double = x => x * 2;                    // Single parameter
const add = (a, b) => a + b;                  // Multiple parameters
const sayHi = () => "Hello!";                 // No parameters
const multiply = (a, b) => {                  // Multi-line
    let result = a * b;
    return result;
};

## 4. PARAMETERS & ARGUMENTS
===========================
// Default Parameters
function multiply(a, b = 1) {
    return a * b;
}
console.log(multiply(5)); // 5 (b uses default 1)

// Rest Parameters (...)
function sum(...numbers) {
    return numbers.reduce((total, num) => total + num, 0);
}
console.log(sum(1, 2, 3, 4)); // 10

## 5. RETURN STATEMENT
======================
function checkAge(age) {
    if (age >= 18) {
        return "Adult";
    } else {
        return "Minor";
    }
    // Code after return won't execute
    console.log("This won't run");
}

## 6. FIRST-CLASS FUNCTIONS (IMPORTANT CONCEPT)
==============================================
// Functions can be:
// 1. Stored in variables
// 2. Passed as arguments
// 3. Returned from other functions
// 4. Stored in data structures

// Example 1: Function as argument
function shout(msg) {
    return msg.toUpperCase();
}

function processMessage(fn) {
    console.log(fn("hello")); // HELLO
}
processMessage(shout); // Passing function as argument

// Example 2: Function returning function
function createMultiplier(multiplier) {
    return function(x) {
        return x * multiplier;
    };
}
const double = createMultiplier(2);
console.log(double(5)); // 10

## 7. HIGHER-ORDER FUNCTIONS
============================
// Functions that take other functions as arguments
// OR return functions

// Array methods are common HOFs:
const numbers = [1, 2, 3, 4, 5];

// forEach
numbers.forEach(num => console.log(num));

// map
const doubled = numbers.map(num => num * 2);

// filter
const evens = numbers.filter(num => num % 2 === 0);

// reduce
const sum = numbers.reduce((total, num) => total + num, 0);

## 8. CALLBACK FUNCTIONS
========================
// Functions passed as arguments to other functions
function calculate(a, b, operation) {
    return operation(a, b);
}

function add(x, y) { return x + y; }
function multiply(x, y) { return x * y; }

console.log(calculate(5, 3, add));       // 8
console.log(calculate(5, 3, multiply));  // 15

## 9. IMMEDIATELY INVOKED FUNCTION EXPRESSIONS (IIFE)
=====================================================
(function() {
    console.log("I run immediately!");
})();

// With parameters:
(function(name) {
    console.log("Hello " + name);
})("Buddy");

## 10. FUNCTION SCOPE
=====================
let globalVar = "I'm global";

function testScope() {
    let localVar = "I'm local";
    console.log(globalVar); // Accessible
    console.log(localVar);  // Accessible
}

testScope();
// console.log(localVar); // Error - localVar not defined outside

## 11. CLOSURES (ADVANCED CONCEPT)
==================================
// Inner function has access to outer function's variables
// even after outer function has returned

function outer() {
    let count = 0;
    
    return function() {
        count++;
        return count;
    };
}

const counter = outer();
console.log(counter()); // 1
console.log(counter()); // 2
console.log(counter()); // 3

## 12. RECURSIVE FUNCTIONS
==========================
// Functions that call themselves

function factorial(n) {
    if (n === 0 || n === 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

console.log(factorial(5)); // 120

## 13. PURE FUNCTIONS
=====================
// Same input always gives same output
// No side effects

// Pure function:
function pureAdd(a, b) {
    return a + b;
}

// Impure function:
let result = 0;
function impureAdd(a, b) {
    result = a + b; // Side effect
    return result;
}

## 14. FUNCTION METHODS
=======================
// call(), apply(), bind()

const person = {
    name: "Buddy",
    greet: function(greeting) {
        return greeting + " " + this.name;
    }
};

// call() - immediate execution with 'this' context
console.log(person.greet.call({name: "Waha"}, "Hello"));

// apply() - similar to call but takes array of arguments
console.log(person.greet.apply({name: "Waha"}, ["Hi"]));

// bind() - returns new function with bound context
const boundGreet = person.greet.bind({name: "Waha"});
console.log(boundGreet("Hey"));

## 15. GENERATOR FUNCTIONS
==========================
function* numberGenerator() {
    yield 1;
    yield 2;
    yield 3;
}

const gen = numberGenerator();
console.log(gen.next().value); // 1
console.log(gen.next().value); // 2
console.log(gen.next().value); // 3

## 16. ASYNC FUNCTIONS
======================
async function fetchData() {
    try {
        const response = await fetch('api-url');
        const data = await response.json();
        return data;
    } catch (error) {
        console.error("Error:", error);
    }
}

## QUICK REFERENCE - FUNCTION TYPES
===================================
1. Regular Function: function name() {}
2. Function Expression: const name = function() {}
3. Arrow Function: const name = () => {}
4. IIFE: (function() {})()
5. Generator: function* name() {}
6. Async: async function name() {}
7. Constructor: function Name() {} (with 'new')
8. Method: obj.method = function() {}
9. Callback: function passed as argument
10. Higher-Order: function that takes/returns functions

## BEST PRACTICES
=================
✓ Use descriptive function names
✓ Keep functions small and focused (Single Responsibility)
✓ Limit parameters (max 3-4)
✓ Use default parameters instead of conditionals
✓ Prefer pure functions when possible
✓ Use arrow functions for callbacks
✓ Document complex functions with comments

## COMMON PATTERNS
==================
// Factory Function
function createUser(name, age) {
    return {
        name,
        age,
        isAdult: function() {
            return this.age >= 18;
        }
    };
}

// Constructor Function
function Person(name, age) {
    this.name = name;
    this.age = age;
}
Person.prototype.greet = function() {
    return "Hello, I'm " + this.name;
};

// Module Pattern
const calculator = (function() {
    function add(a, b) { return a + b; }
    function multiply(a, b) { return a * b; }
    
    return {
        add,
        multiply
    };
})();
