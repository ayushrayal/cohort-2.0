# JAVASCRIPT OBJECTS - COMPLETE GUIDE
# ===================================

## 1. OBJECT FUNDAMENTALS
## =======================

### What are Objects?
Objects are collections of key-value pairs that represent real-world entities. They store properties (data) and methods (functions) that define characteristics and behaviors.

### Key Characteristics:
- UNORDERED collection of properties
- KEY-VALUE pairs (keys are strings/Symbols)
- REFERENCE type (passed by reference, not value)
- DYNAMIC (properties can be added/removed at runtime)
- Can contain any data type (including other objects/functions)

### Real-world Analogy:
Think of a "person" object:
- Properties: name, age, email (characteristics)
- Methods: walk(), talk(), eat() (behaviors)

## 2. OBJECT CREATION METHODS
## ==========================

### 1. Object Literal (Most Common):
let person = {
    name: "John",
    age: 30,
    city: "New York"
};

### 2. Constructor Function:
let person = new Object();
person.name = "John";
person.age = 30;

### 3. Object.create():
let prototypeObj = { type: "human" };
let person = Object.create(prototypeObj);
person.name = "John";

### 4. Class Syntax (ES6):
class Person {
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
}
let person = new Person("John", 30);

## 3. PROPERTY ACCESS METHODS
## ==========================

### Dot Notation:
let person = { name: "John", age: 30 };
console.log(person.name);  // "John"
console.log(person.age);   // 30

### Bracket Notation:
console.log(person["name"]); // "John"
let key = "age";
console.log(person[key]);    // 30

### When to Use Which:

USE DOT NOTATION WHEN:
- Property name is known at coding time
- Property name is valid identifier (no spaces/special chars)
- Simple property access

USE BRACKET NOTATION WHEN:
- Property name is dynamic (stored in variable)
- Property name contains spaces/special characters
- Property name starts with number

### Examples:
let person = {
    "first name": "John",
    "last name": "Doe",
    "1stProperty": "value"
};

console.log(person["first name"]);  // Required for spaces
console.log(person["1stProperty"]); // Required for numbers

let propertyName = "last name";
console.log(person[propertyName]);  // Dynamic access

## 4. OBJECT METHODS
## =================

### What are Methods?
Methods are functions that are properties of objects. They define the object's behavior.

### Method Definitions:
let person = {
    name: "John",
    
    // Traditional method
    greet: function() {
        return `Hello, I'm ${this.name}`;
    },
    
    // ES6 shorthand
    introduce() {
        return `Hi, I'm ${this.name}`;
    },
    
    // Arrow function (careful with 'this')
    sayHello: () => {
        console.log("Hello!");
    }
};

### Method Invocation:
console.log(person.greet());     // "Hello, I'm John"
console.log(person.introduce()); // "Hi, I'm John"

## 5. THE 'this' KEYWORD
## =====================

### What is 'this'?
'this' refers to the object that is currently executing the code. Its value depends on how a function is called.

### 'this' Binding Rules:

1. METHOD INVOCATION:
   - 'this' refers to the object
   let obj = {
       name: "John",
       greet() { console.log(this.name); } // 'this' = obj
   };

2. FUNCTION INVOCATION:
   - 'this' refers to global object (window/global)
   - undefined in strict mode
   function test() { console.log(this); }

3. CONSTRUCTOR INVOCATION:
   - 'this' refers to newly created instance
   function Person(name) { this.name = name; }

4. EXPLICIT BINDING:
   - 'this' is explicitly set using call/apply/bind
   func.call(context), func.apply(context), func.bind(context)

### Examples:
let person = {
    name: "John",
    greet: function() {
        console.log(`Hello, ${this.name}`);
    }
};

person.greet(); // "Hello, John" - 'this' refers to person

let greetFunc = person.greet;
greetFunc(); // "Hello, undefined" - 'this' lost context

## 6. OBJECT ITERATION
## ===================

### for...in Loop:
let person = { name: "John", age: 30, city: "NYC" };

for (let key in person) {
    console.log(`${key}: ${person[key]}`);
}
// name: John
// age: 30  
// city: NYC

### Object.keys():
let keys = Object.keys(person); // ["name", "age", "city"]
keys.forEach(key => {
    console.log(key, person[key]);
});

### Object.values():
let values = Object.values(person); // ["John", 30, "NYC"]

### Object.entries():
let entries = Object.entries(person); 
// [["name", "John"], ["age", 30], ["city", "NYC"]]

entries.forEach(([key, value]) => {
    console.log(key, value);
});

## 7. OBJECT BUILT-IN METHODS
## ==========================

### Object.assign() - Copy/Merge:
let target = { a: 1, b: 2 };
let source = { b: 3, c: 4 };

let result = Object.assign(target, source);
// { a: 1, b: 3, c: 4 } - target is modified

// Better: Create new object
let copy = Object.assign({}, target, source);

### Object.freeze() - Immutable:
let obj = { name: "John" };
Object.freeze(obj);
obj.name = "Jane"; // Fails silently in non-strict mode
console.log(obj.name); // "John"

### Object.seal() - No Add/Remove:
let obj = { name: "John" };
Object.seal(obj);
obj.name = "Jane";     // Allowed
obj.age = 30;          // Not allowed
delete obj.name;       // Not allowed

### Object.hasOwnProperty():
let obj = { name: "John" };
console.log(obj.hasOwnProperty("name")); // true
console.log(obj.hasOwnProperty("toString")); // false (inherited)

## 8. CONSTRUCTOR FUNCTIONS
## ========================

### Theory:
Constructor functions act as blueprints for creating multiple objects with the same structure.

### Convention:
- Function name in PascalCase
- Use 'new' keyword to create instances
- 'this' refers to the new instance

### Example:
function Person(name, age, city) {
    // Properties
    this.name = name;
    this.age = age;
    this.city = city;
    
    // Methods (created for each instance)
    this.introduce = function() {
        return `Hi, I'm ${this.name} from ${this.city}`;
    };
}

// Creating instances
let john = new Person("John", 30, "New York");
let jane = new Person("Jane", 25, "London");

console.log(john.introduce()); // "Hi, I'm John from New York"

## 9. ES6 CLASSES
## ==============

### Theory:
Classes are syntactic sugar over constructor functions, providing cleaner syntax and built-in inheritance.

### Class Syntax:
class Person {
    // Constructor
    constructor(name, age, city) {
        this.name = name;
        this.age = age;
        this.city = city;
    }
    
    // Instance method (added to prototype)
    introduce() {
        return `Hi, I'm ${this.name} from ${this.city}`;
    }
    
    // Static method (called on class, not instances)
    static species() {
        return "Homo sapiens";
    }
    
    // Getter
    get description() {
        return `${this.name}, ${this.age} years old`;
    }
    
    // Setter
    set updateAge(newAge) {
        if (newAge > 0) {
            this.age = newAge;
        }
    }
}

### Usage:
let john = new Person("John", 30, "New York");
console.log(john.introduce()); // Instance method
console.log(Person.species()); // Static method
console.log(john.description); // Getter
john.updateAge = 31;          // Setter

## 10. OBJECT DESTRUCTURING
## ========================

### Theory:
Destructuring allows extracting multiple properties from objects into distinct variables.

### Basic Destructuring:
let person = { name: "John", age: 30, city: "NYC" };

let { name, age } = person;
console.log(name, age); // "John" 30

### Renaming Variables:
let { name: personName, city: personCity } = person;
console.log(personName, personCity); // "John" "NYC"

### Default Values:
let { name, salary = 50000 } = person;
console.log(salary); // 50000 (default)

### Rest Operator:
let { name, ...rest } = person;
console.log(rest); // { age: 30, city: "NYC" }

### Nested Destructuring:
let company = {
    name: "Tech Corp",
    address: {
        street: "123 Main St",
        city: "Boston"
    }
};

let { name, address: { city } } = company;
console.log(city); // "Boston"

## 11. SPREAD OPERATOR WITH OBJECTS
## ================================

### Copying Objects:
let original = { a: 1, b: 2 };
let copy = { ...original }; // Shallow copy

### Merging Objects:
let obj1 = { a: 1, b: 2 };
let obj2 = { b: 3, c: 4 };
let merged = { ...obj1, ...obj2 }; // { a: 1, b: 3, c: 4 }

### Overriding Properties:
let person = { name: "John", age: 30 };
let updated = { ...person, age: 31 }; // { name: "John", age: 31 }

### Adding Properties:
let newPerson = { ...person, salary: 50000 };

## 12. OBJECT PROTOTYPES
## =====================

### Prototype Chain:
Every JavaScript object has a prototype from which it inherits properties and methods.

### Adding to Prototype:
function Person(name) {
    this.name = name;
}

// Method added to prototype (shared across instances)
Person.prototype.greet = function() {
    return `Hello, I'm ${this.name}`;
};

let john = new Person("John");
console.log(john.greet()); // "Hello, I'm John"

### Prototype Inheritance:
console.log(john.toString()); // Inherited from Object.prototype
console.log(john.hasOwnProperty("name")); // Inherited method

## 13. PROPERTY DESCRIPTORS
## ========================

### Getting Descriptors:
let obj = { name: "John" };
let descriptor = Object.getOwnPropertyDescriptor(obj, "name");

console.log(descriptor);
// {
//   value: "John",
//   writable: true,
//   enumerable: true, 
//   configurable: true
// }

### Setting Descriptors:
Object.defineProperty(obj, "age", {
    value: 30,
    writable: false,      // Cannot change
    enumerable: true,     // Shows up in loops
    configurable: false   // Cannot delete or reconfigure
});

obj.age = 40; // Fails silently (non-strict mode)
console.log(obj.age); // 30

## 14. REAL-WORLD OBJECT PATTERNS
## ===============================

### Factory Function Pattern:
function createUser(name, email, role = "user") {
    return {
        name,
        email,
        role,
        isActive: true,
        createdAt: new Date(),
        
        activate() {
            this.isActive = true;
        },
        
        deactivate() {
            this.isActive = false;
        },
        
        getInfo() {
            return `${this.name} (${this.role}) - ${this.email}`;
        }
    };
}

let user1 = createUser("John", "john@email.com", "admin");
let user2 = createUser("Jane", "jane@email.com");

### Module Pattern:
const UserManager = (function() {
    let users = [];
    let nextId = 1;
    
    return {
        addUser(name, email) {
            const user = {
                id: nextId++,
                name,
                email,
                createdAt: new Date()
            };
            users.push(user);
            return user;
        },
        
        getUser(id) {
            return users.find(user => user.id === id);
        },
        
        getAllUsers() {
            return [...users]; // Return copy
        },
        
        deleteUser(id) {
            users = users.filter(user => user.id !== id);
        }
    };
})();

// Usage
UserManager.addUser("John", "john@email.com");
console.log(UserManager.getAllUsers());

## 15. COMMON OBJECT OPERATIONS
## ============================

### Property Existence:
let obj = { name: "John", age: 30 };

console.log("name" in obj);              // true
console.log(obj.hasOwnProperty("name")); // true
console.log(obj.address);                // undefined

### Safe Navigation (Optional Chaining - ES2020):
console.log(obj?.address?.city);         // undefined (no error)
console.log(obj?.getName?.());           // Safe method call

### Property Enumeration:
let obj = { name: "John", age: 30 };

// Get own property names (including non-enumerable)
let allProps = Object.getOwnPropertyNames(obj);

// Get property descriptors
let descriptors = Object.getOwnPropertyDescriptors(obj);

### Cloning Objects:
// Shallow clone
let shallowClone = { ...obj };

// Deep clone (for nested objects)
let deepClone = JSON.parse(JSON.stringify(obj));

// Using Object.assign()
let clone = Object.assign({}, obj);

## 16. OBJECT VS ARRAY: WHEN TO USE
## ================================

### Use Objects When:
// Dealing with key-value pairs
let user = { 
    id: 1, 
    name: "John", 
    email: "john@email.com" 
};

// Need named properties
let config = { 
    theme: "dark", 
    language: "en", 
    notifications: true 
};

// Representing real-world entities
let product = { 
    name: "Laptop", 
    price: 1000, 
    category: "Electronics" 
};

### Use Arrays When:
// Dealing with ordered lists
let numbers = [1, 2, 3, 4, 5];

// Need index-based access
let users = ["John", "Jane", "Bob"];

// Sequential data
let steps = ["Step 1", "Step 2", "Step 3"];

## 17. BEST PRACTICES
## ==================

### Naming Conventions:
- Use camelCase for properties/methods
- Use descriptive names
- Be consistent with naming patterns

### Structure:
- Group related properties together
- Use methods for actions/behaviors
- Keep objects focused (Single Responsibility)

### Performance:
- Avoid deep nesting when possible
- Use object pooling for frequently created objects
- Consider using Maps for frequent add/remove operations

### Maintenance:
- Use destructuring for clean code
- Leverage spread operator for immutability
- Document complex object structures

## 18. COMMON PITFALLS
## ===================

### 1. Mutating Original Objects:
let original = { a: 1, b: 2 };
let copy = original; // Reference, not copy!
copy.a = 99;
console.log(original.a); // 99 (unintended mutation)

// Solution: Always create copies
let safeCopy = { ...original };

### 2. 'this' Context Loss:
let obj = {
    name: "John",
    greet: function() {
        console.log(this.name);
    }
};

let greetFunc = obj.greet;
greetFunc(); // 'this' is undefined/lost

// Solution: Use bind or arrow functions
let boundGreet = obj.greet.bind(obj);

### 3. Shallow vs Deep Copy:
let original = { 
    a: 1, 
    b: { 
        c: 2 
    } 
};

let shallowCopy = { ...original };
shallowCopy.b.c = 99;
console.log(original.b.c); // 99 (nested object shared)

// Solution: Deep copy needed for nested objects
let deepCopy = JSON.parse(JSON.stringify(original));
